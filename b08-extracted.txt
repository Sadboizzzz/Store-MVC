using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.ComponentModel.DataAnnotations;
namespace DBSportStore.Models
{
    // =========================================================
    // VIEWMODEL DÙNG CHO FORM SIGN UP
    // =========================================================
    // SignUpViewModel không phải bảng CSDL.
    // Nó chỉ gom các dữ liệu người dùng nhập trên form Đăng ký.
    public class SignUpViewModel
    {
        // Bắt buộc nhập UserName.
        [Required(
            ErrorMessage = "Vui lòng nhập tên đăng nhập.")]
        // UserAccount.UserName trong CSDL có độ dài tối đa 50.
        [StringLength(50)]
        // Tên hiển thị trên Label.
        [Display(Name = "Tên đăng nhập")]
        public string UserName { get; set; }
        // Bắt buộc nhập Password.
        [Required(
            ErrorMessage = "Vui lòng nhập mật khẩu.")]
        // Trong bài này yêu cầu ít nhất 6 ký tự.
        [StringLength(
            50,
            MinimumLength = 6,
            ErrorMessage = "Mật khẩu phải có ít nhất 6 ký tự.")]
        // View sẽ hiển thị ô nhập dạng Password.
        [DataType(DataType.Password)]
        [Display(Name = "Mật khẩu")]
        public string Password { get; set; }
        // Người dùng phải nhập lại Password.
        [Required(
            ErrorMessage = "Vui lòng nhập lại mật khẩu.")]
        [DataType(DataType.Password)]
        // Compare kiểm tra ConfirmPassword
        // phải giống với property Password ở phía trên.
        [Compare(
            "Password",
            ErrorMessage = "Mật khẩu nhập lại không khớp.")]
        [Display(Name = "Nhập lại mật khẩu")]
        public string ConfirmPassword { get; set; }
        [Required(
            ErrorMessage = "Vui lòng nhập họ tên.")]
        [StringLength(100)]
        [Display(Name = "Họ tên")]
        public string FullName { get; set; }
        [Required(
            ErrorMessage = "Vui lòng nhập số điện thoại.")]
        [StringLength(20)]
        [Display(Name = "Số điện thoại")]
        public string Phone { get; set; }
        [Required(
            ErrorMessage = "Vui lòng nhập email.")]
        // Kiểm tra định dạng email cơ bản.
        [EmailAddress(
            ErrorMessage = "Email không đúng định dạng.")]
        [StringLength(100)]
        [Display(Name = "Email")]
        public string Email { get; set; }
        [Required(
            ErrorMessage = "Vui lòng nhập địa chỉ.")]
        [StringLength(250)]
        [Display(Name = "Địa chỉ")]
        public string Address { get; set; }
    }
}

        {

                db.Dispose();

        }

MỤC TIÊU

3.1. Tạo Models/PasswordHelper.cs
Bước 1. Click phải Models → Add → Class...
Bước 2. Đặt tên PasswordHelper.cs.
Models/PasswordHelper.cs
using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Security.Cryptography;
namespace DBSportStore.Models
{
    // =========================================================
    // HELPER XỬ LÝ PASSWORD CHO BÀI THỰC HÀNH
    // =========================================================
    public static class PasswordHelper
    {
        // =====================================================
        // TẠO SALT NGẪU NHIÊN
        // =====================================================
        public static string CreateSalt()
        {
            // Tạo mảng 16 byte.
            // Mỗi tài khoản sẽ có Salt khác nhau.
            byte[] saltBytes =
                new byte[16];
            // RandomNumberGenerator tạo dữ liệu ngẫu nhiên
            // phù hợp hơn Random thông thường cho mục đích bảo mật.
            using (var rng =
                RandomNumberGenerator.Create())
            {
                // Ghi dữ liệu ngẫu nhiên vào saltBytes.
                rng.GetBytes(saltBytes);
            }
            // CSDL lưu chuỗi, nên chuyển byte[]
            // thành chuỗi Base64 trước khi trả về.
            return Convert.ToBase64String(
                saltBytes);
        }
        // =====================================================
        // HASH PASSWORD
        // =====================================================
        public static string Hash(
            string password,
            string salt)
        {
            // Chuyển Salt từ chuỗi Base64 trong CSDL
            // trở lại thành byte[].
            byte[] saltBytes =
                Convert.FromBase64String(salt);
            // Rfc2898DeriveBytes triển khai PBKDF2.
            //
            // password:
            // mật khẩu người dùng nhập.
            //
            // saltBytes:
            // Salt riêng của tài khoản.
            //
            // 100000:
            // số vòng lặp dùng để làm việc dò mật khẩu chậm hơn.
            using (var pbkdf2 =
                new Rfc2898DeriveBytes(
                    password,
                    saltBytes,
                    100000))
            {
                // Lấy 32 byte kết quả hash.
                byte[] hashBytes =
                    pbkdf2.GetBytes(32);
                // Chuyển hash thành Base64
                // để lưu vào UserAccount.PasswordHash.
                return Convert.ToBase64String(
                    hashBytes);
            }
        }
        // =====================================================
        // KIỂM TRA PASSWORD KHI LOGIN
        // =====================================================
        public static bool Verify(
            string password,
            string salt,
            string storedHash)
        {
            // Hash lại Password người dùng vừa nhập
            // bằng đúng Salt đã lưu trong CSDL.
            string newHash =
                Hash(password, salt);
            // Hai chuỗi hash giống nhau
            // nghĩa là Password nhập vào đúng.
            return newHash == storedHash;
        }
    }
}

// =====================================================
// SIGN UP - TẠO USERACCOUNT VÀ CUSTOMER
// =====================================================
[HttpPost]
[ValidateAntiForgeryToken]
public ActionResult SignUp(
    SignUpViewModel model)
{
    // Dữ liệu form phải hợp lệ trước khi xử lý CSDL.
    if (!ModelState.IsValid)
    {
        return View(model);
    }
    // =================================================
    // 1. KIỂM TRA USERNAME
    // =================================================
    // Any() trả true nếu đã có UserAccount
    // sử dụng đúng UserName người dùng đang nhập.
    bool userNameExists =
        db.UserAccount.Any(
            u => u.UserName == model.UserName);
    if (userNameExists)
    {
        // Gắn lỗi trực tiếp vào ô UserName.
        ModelState.AddModelError(
            "UserName",
            "Tên đăng nhập đã tồn tại.");
        return View(model);
    }
    // =================================================
    // 2. LẤY ROLE CUSTOMER
    // =================================================
    // Tài khoản đăng ký công khai luôn là Customer.
    // Người dùng KHÔNG được chọn Role từ form.
    var customerRole =
        db.Role.FirstOrDefault(
            r => r.RoleName == "Customer");
    if (customerRole == null)
    {
        ModelState.AddModelError(
            "",
            "Không tìm thấy Role Customer.");
        return View(model);
    }
    // =================================================
    // 3. TẠO SALT + PASSWORD HASH
    // =================================================
    // Mỗi tài khoản tạo một Salt mới.
    string salt =
        PasswordHelper.CreateSalt();
    // PasswordHash được tạo từ Password + Salt.
    string passwordHash =
        PasswordHelper.Hash(
            model.Password,
            salt);
    // =================================================
    // 4. BẮT ĐẦU TRANSACTION
    // =================================================
    // UserAccount và Customer phải cùng được tạo thành công.
    // Nếu một bước lỗi -> rollback toàn bộ.
    using (var transaction =
        db.Database.BeginTransaction())
    {
        try
        {
            // =============================================
            // 4.1. TẠO USERACCOUNT
            // =============================================
            var account = new UserAccount
            {
                UserName = model.UserName,
                // Không lưu Password dạng rõ.
                PasswordHash = passwordHash,
                PasswordSalt = salt,
                // Gán Role Customer lấy từ CSDL.
                RoleID = customerRole.RoleID,
                IsActive = true,
                // Ghi thời điểm tạo tài khoản.
                CreatedAt = DateTime.Now
            };
            // Đánh dấu entity cần INSERT.
            db.UserAccount.Add(account);
            // Ghi UserAccount xuống CSDL.
            db.SaveChanges();
            // Sau SaveChanges(), SQL Server sinh UserID.
            // EF tự cập nhật giá trị vào account.UserID.
            // =============================================
            // 4.2. TẠO CUSTOMER GẮN VỚI USERID VỪA SINH
            // =============================================
            var customer = new Customer
            {
                // Đây là UserID do SQL Server sinh.
                // KHÔNG lấy UserID từ dropdown hoặc từ form.
                UserID = account.UserID,
                FullName = model.FullName,
                Phone = model.Phone,
                Email = model.Email,
                Address = model.Address
            };
            db.Customer.Add(customer);
            db.SaveChanges();
            // Cả UserAccount và Customer đều thành công.
            transaction.Commit();
            TempData["SignUpSuccess"] =
                "Đăng ký thành công. "
                + "Vui lòng đăng nhập.";
            // Chức năng 3 sẽ tạo trang Login hoàn chỉnh.
            return RedirectToAction("Login");
        }
        catch
        {
            // Có lỗi -> hủy toàn bộ thay đổi
            // trong transaction.
            transaction.Rollback();
            ModelState.AddModelError(
                "",
                "Có lỗi khi tạo tài khoản. "
                + "Vui lòng thử lại.");
            return View(model);
        }
    }
}

// =====================================================
// LOGIN
// =====================================================
[HttpPost]
[ValidateAntiForgeryToken]
public ActionResult Login(
    LoginViewModel model)
{
    // Validation form phải hợp lệ.
    if (!ModelState.IsValid)
    {
        return View(model);
    }
    // =================================================
    // 1. TÌM USERACCOUNT
    // =================================================
    // Include(Role) để lấy thêm RoleName
    // của tài khoản trong cùng truy vấn.
    var account = db.UserAccount
        .Include(u => u.Role)
        .FirstOrDefault(
            u => u.UserName == model.UserName
              && u.IsActive);
    // Không tìm thấy User hoặc tài khoản bị khóa.
    if (account == null)
    {
        ModelState.AddModelError(
            "",
            "Tên đăng nhập hoặc mật khẩu không đúng.");
        return View(model);
    }
    // =================================================
    // 2. KIỂM TRA PASSWORD
    // =================================================
    // Verify():
    // - lấy Password người dùng vừa nhập
    // - Hash lại bằng PasswordSalt trong CSDL
    // - so sánh với PasswordHash đã lưu.
    bool passwordCorrect =
        PasswordHelper.Verify(
            model.Password,
            account.PasswordSalt,
            account.PasswordHash);
    if (!passwordCorrect)
    {
        ModelState.AddModelError(
            "",
            "Tên đăng nhập hoặc mật khẩu không đúng.");
        return View(model);
    }
    // =================================================
    // 3. LƯU TRẠNG THÁI LOGIN VÀO SESSION
    // =================================================
    // Session thuộc riêng phiên làm việc hiện tại.
    // Các trang khác có thể đọc lại những giá trị này.
    Session["UserID"] =
        account.UserID;
    Session["UserName"] =
        account.UserName;
    // RoleName có giá trị Admin hoặc Customer.
    Session["Role"] =
        account.Role.RoleName;
    // =================================================
    // 4. NẾU LÀ CUSTOMER -> LẤY HỒ SƠ CUSTOMER
    // =================================================
    var customer = db.Customer
        .FirstOrDefault(
            c => c.UserID == account.UserID);
    if (customer != null)
    {
        Session["CustomerID"] =
            customer.CustomerID;
        Session["FullName"] =
            customer.FullName;
    }
    else
    {
        // Admin có thể không có Customer.
        Session.Remove("CustomerID");
        Session.Remove("FullName");
    }
    // =================================================
    // 5. CHUYỂN TRANG SAU LOGIN
    // =================================================
    // Admin vào trang quản trị.
    if (account.Role.RoleName == "Admin")
    {
        return RedirectToAction(
            "Index",
            "Admin");
    }
    // Customer quay về Trang Chủ.
    return RedirectToAction(
        "Index",
        "Store");
}

 public ActionResult CreateAdminForLab()
 {

         db.UserAccount.Any(

         PasswordHelper.CreateSalt();
     var admin =

             PasswordSalt = salt,
             // Password lab: Admin123
             PasswordHash =
                 PasswordHelper.Hash(

             RoleID = adminRole.RoleID,
             IsActive = true,
             CreatedAt = DateTime.Now

     db.UserAccount.Add(admin);
     db.SaveChanges();

Filters/AdminAuthorizeAttribute.cs
using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Web.Mvc;
using System.Web.Routing;
namespace DBSportStore.Filters
{
    // =========================================================
    // ATTRIBUTE KIỂM TRA QUYỀN ADMIN
    // =========================================================
    // Kế thừa AuthorizeAttribute để có thể gắn
    // [AdminAuthorize] lên Controller hoặc Action.
    public class AdminAuthorizeAttribute
        : AuthorizeAttribute
    {
        // =====================================================
        // KIỂM TRA REQUEST CÓ PHẢI ADMIN HAY KHÔNG
        // =====================================================
        protected override bool AuthorizeCore(
            HttpContextBase httpContext)
        {
            // Chưa có UserID trong Session
            // nghĩa là chưa Login.
            if (httpContext.Session["UserID"] == null)
            {
                return false;
            }
            // Convert.ToString() tránh lỗi nếu Session["Role"] null.
            string role =
                Convert.ToString(
                    httpContext.Session["Role"]);
            // Chỉ Role Admin mới trả true.
            return role == "Admin";
        }
        // =====================================================
        // XỬ LÝ KHI KHÔNG CÓ QUYỀN
        // =====================================================
        protected override void HandleUnauthorizedRequest(
            AuthorizationContext filterContext)
        {
            // Chưa Login -> chuyển về trang Login.
            if (filterContext.HttpContext
                .Session["UserID"] == null)
            {
                filterContext.Result =
                    new RedirectToRouteResult(
                        new RouteValueDictionary(
                            new
                            {
                                controller = "Account",
                                action = "Login"
                            }));
                return;
            }
            // Đã Login nhưng không phải Admin
            // -> trả về HTTP 403 Forbidden.
            filterContext.Result =
                new HttpStatusCodeResult(
                    403,
                    "Bạn không có quyền truy cập trang quản trị.");
        }
    }
}

// =====================================================
// MỞ FORM CHECKOUT
// =====================================================
public ActionResult Checkout()
{
    // Lấy Cart hiện tại từ Session.
    var cart = GetCart();
    // Không cho Checkout nếu giỏ trống.
    if (!cart.Items.Any())
    {
        return RedirectToAction("ShowCart");
    }
    // Tạo ViewModel cho trang Checkout.
    var model = new CheckoutViewModel
    {
        Cart = cart
    };
    // =================================================
    // NẾU CUSTOMER ĐÃ LOGIN -> ĐIỀN SẴN HỒ SƠ
    // =================================================
    // CustomerID chỉ có khi Login bằng tài khoản
    // đã gắn với một Customer.
    if (Session["CustomerID"] != null)
    {
        // Session lưu object nên ép về int.
        int customerId =
            (int)Session["CustomerID"];
        var customer =
            db.Customer.Find(customerId);
        if (customer != null)
        {
            model.FullName =
                customer.FullName;
            model.Phone =
                customer.Phone;
            model.Email =
                customer.Email;
            model.Address =
                customer.Address;
            // Mặc định địa chỉ giao hàng
            // bằng địa chỉ hồ sơ Customer.
            model.ShippingAddress =
                customer.Address;
        }
    }
    return View(model);
}
